# Applied to every install in install.sh.
#
# motorbridge must be the motorbridge_2ec fork (non-blocking encoder reads). Its version
# exists only on the fork's GitHub releases, never on PyPI, so pinning it exactly means no
# dependency can ever replace it with PyPI's motorbridge 0.5.6. Bump together with
# motorbridge_2ec/bindings/python/src/motorbridge/_version.py and the fork's 2ec-<version> tag.
motorbridge==0.5.6+2ec.1

# On the robot (Linux aarch64), PyPI's torch is a CUDA build that drags in ~3.6 GB of
# nvidia-* and triton packages a Raspberry Pi cannot use. Use PyTorch's CPU-only wheels
# (install.sh adds their index on that platform). Bump these together with LeRobot's torch
# bounds (lerobot 0.6.1: torch>=2.7,<2.12, torchvision>=0.22,<0.27).
torch==2.11.0+cpu; sys_platform == "linux" and platform_machine == "aarch64"
torchvision==0.26.0+cpu; sys_platform == "linux" and platform_machine == "aarch64"

# Orbbec camera SDK (Gemini 336L). install.sh installs it with --no-deps: the SDK itself needs
# only numpy, and its declared dependencies are for its examples (pynput -> evdev compiles
# from source; opencv-python would overwrite lerobot's opencv-python-headless cv2).
pyorbbecsdk2==2.1.3
